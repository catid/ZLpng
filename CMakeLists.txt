cmake_minimum_required(VERSION 3.20.2)
project(ZLpng VERSION 0.1.0 LANGUAGES C CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "OpenZL requires a 64-bit target")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(ZLPNG_NATIVE "Optimize for the build machine's CPU" OFF)
option(ZLPNG_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(ZLPNG_BUILD_TESTS "Build codec tests" ON)
option(ZLPNG_BUILD_BENCHMARKS "Build Linux corpus benchmark and isolated original control" OFF)
option(ZLPNG_BUILD_LEGACY "Build the original ZPNG tools" OFF)
option(ZPNG_BUILD_EXPERIMENTS "Build earlier Zstd filter experiment tools" OFF)
if(ZLPNG_NATIVE AND NOT MSVC)
    add_compile_options(-march=native)
endif()
if(ZLPNG_SANITIZE AND NOT MSVC)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
include(cmake/OpenZL.cmake)

add_library(zlpng_backend STATIC zlpng_backend.c)
target_include_directories(zlpng_backend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(zlpng_backend PUBLIC openzl)
add_library(zlpnglib STATIC zlpng.cpp)
add_library(ZLpng::ZLpng ALIAS zlpnglib)
target_include_directories(zlpnglib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(zlpnglib PUBLIC zlpng_backend)
add_executable(zlpng apps/zlpng_cli.cpp)
target_link_libraries(zlpng PRIVATE zlpnglib)

if(ZLPNG_BUILD_TESTS)
    enable_testing()
    add_executable(zlpng_test tests/test_zlpng.cpp)
    target_link_libraries(zlpng_test PRIVATE zlpnglib)
    add_test(NAME zlpng_api COMMAND zlpng_test)
    add_executable(zlpng_backend_test tests/test_zlpng_backend.c)
    target_link_libraries(zlpng_backend_test PRIVATE zlpng_backend)
    add_test(NAME zlpng_backend COMMAND zlpng_backend_test)
    add_executable(zlpng_c_test tests/test_zlpng_c.c)
    target_link_libraries(zlpng_c_test PRIVATE zlpnglib)
    add_test(NAME zlpng_c_api COMMAND zlpng_c_test)
endif()

if(ZLPNG_BUILD_LEGACY OR ZPNG_BUILD_EXPERIMENTS OR ZLPNG_BUILD_BENCHMARKS)
    include(cmake/Legacy.cmake)
endif()
if(ZLPNG_BUILD_BENCHMARKS)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The paired benchmark uses Linux dlopen and CPU affinity")
    endif()
    add_library(zpng_reference SHARED ${ZPNG_LIB_SRCFILES} ${ZSTD_LIB_SRCFILES})
    target_link_libraries(zpng_reference PRIVATE Threads::Threads)
    target_link_options(zpng_reference PRIVATE -Wl,-Bsymbolic
        -Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/cmake/zpng_reference.map)
    add_executable(zlpng_bench experiments/round6/zlpng_bench.cpp)
    target_link_libraries(zlpng_bench PRIVATE zlpnglib ${CMAKE_DL_LIBS})
endif()
