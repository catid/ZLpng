cmake_minimum_required(VERSION 3.20.2)
project(ZLpng VERSION 0.2.0 LANGUAGES C CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "OpenZL requires a 64-bit target")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(ZLPNG_NATIVE "Optimize for the build machine's CPU" OFF)
option(ZLPNG_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(ZLPNG_BUILD_TESTS "Build codec tests" ON)
option(ZLPNG_BUILD_BENCHMARKS "Build Linux corpus benchmark" OFF)
if(ZLPNG_NATIVE AND NOT MSVC)
    add_compile_options(-march=native)
endif()
if(ZLPNG_SANITIZE AND NOT MSVC)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
include(cmake/OpenZL.cmake)

add_library(zlpng_backend STATIC zlpng_backend.c)
target_include_directories(zlpng_backend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(zlpng_backend PUBLIC openzl)
add_library(zlpnglib STATIC zlpng.cpp)
add_library(ZLpng::ZLpng ALIAS zlpnglib)
target_include_directories(zlpnglib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(zlpnglib PUBLIC zlpng_backend Threads::Threads)
add_executable(zlpng apps/zlpng_cli.cpp)
target_link_libraries(zlpng PRIVATE zlpnglib)

if(ZLPNG_BUILD_TESTS)
    enable_testing()
    add_executable(zlpng_filter_test tests/test_med_filter.cpp)
    add_test(NAME zlpng_filter COMMAND zlpng_filter_test)
    add_executable(zlpng_test tests/test_zlpng.cpp)
    target_link_libraries(zlpng_test PRIVATE zlpnglib)
    add_test(NAME zlpng_api COMMAND zlpng_test)
    add_executable(zlpng_backend_test tests/test_zlpng_backend.c)
    target_link_libraries(zlpng_backend_test PRIVATE zlpng_backend)
    add_test(NAME zlpng_backend COMMAND zlpng_backend_test)
    add_executable(zlpng_c_test tests/test_zlpng_c.c)
    target_link_libraries(zlpng_c_test PRIVATE zlpnglib)
    add_test(NAME zlpng_c_api COMMAND zlpng_c_test)
endif()

if(ZLPNG_BUILD_BENCHMARKS)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The benchmark uses Linux dlopen and CPU affinity")
    endif()
    include(cmake/Benchmark.cmake)
    find_package(PNG REQUIRED)
    add_executable(zlpng_bench benchmarks/bench.cpp)
    target_link_libraries(zlpng_bench PRIVATE zlpnglib PNG::PNG ${CMAKE_DL_LIBS})
endif()
